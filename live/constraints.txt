# Versions qualified together for the maintained web/kiosk/terminal runtime.
# striqt remains pinned by commit in setup.sh. Python's Requires-Python metadata
# selects compatible transitive releases on supported Python 3.9+ hosts.
fastapi==0.115.14
uvicorn==0.34.3
# NumPy is pinned per interpreter. 2.0.2 is the last release that supports
# Python 3.9 (the AIR-T's interpreter) and it publishes no cp313 wheels — on a
# Debian 13 / Raspberry Pi OS Trixie host (Python 3.13) pip would fall back to
# a source build that cannot compile against the 3.13 C API. 2.1.3 covers
# 3.10-3.13. Both satisfy numba's numpy<2.2 bound inside the striqt stack.
numpy==2.0.2; python_version < "3.13"
numpy==2.1.3; python_version >= "3.13"
